# ─────────────────────────────────────────────
# WIND-TURBINE
# ─────────────────────────────────────────────

# --- MQTT ---
[Unit]
Description=MQTT Wind Turbine Generator
After=network.target

[Service]
Type=simple
User=root
WorkingDirectory=/root/Sample-Data-Generator
ExecStart=/root/Sample-Data-Generator/venv/bin/python3 /root/Sample-Data-Generator/source/main.py \
  wind-turbine \
  --method mqtt \
  --conn anyloguser:mqtt4AnyLog!@172.104.228.251:1883 \
  --db-name wind_turbine \
  --wait-time 1 \
  --iterations 0
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target
